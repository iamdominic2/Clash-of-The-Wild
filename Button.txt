//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    STICK ACTION = 0;
    Motion.GoToXY(-180, -140);
    Looks.SwitchCostumeTo(costume1);
    Looks.SetEffectTo(GHOST, 30);
    Control.CreateCloneOf(_myself_);
}


//----- Key pressed events --------------------------------------------------------------------------------------------

WhenKeyPressed(any)
{
    Looks.SetEffectTo(GHOST, 100);
}

{

}
