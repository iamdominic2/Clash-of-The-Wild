//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    Looks.Hide();
}


//----- Start as clone ------------------------------------------------------------------------------------------------

Control.WhenIStartAsClone()
{
    Motion.GoTo(Player);
    Motion.PointInDirection(Sensing.Of(direction, Player));
    Motion.SetRotationStyle(all around);
    Looks.Show();
    Looks.SwitchCostumeTo(Sword);
    Looks.SetSizeTo(200);
    Motion.Move(3);
    Repeat (7)
    {
        Motion.Move(7.5);
        Looks.ChangeEffectBy(GHOST, 15);
        If Sensing.TouchingObject(Wall)
        {
            Control.Wait(0);
            Control.DeleteThisClone();
        }
    }
    Control.DeleteThisClone();
}


//----- Custom blocks -------------------------------------------------------------------------------------------------

Define Fix collision (warp=true)
{
    Motion.PointInDirection(90);
    Freeze = 1;
    Repeat Until (Not (Sensing.TouchingObject(Wall)))
    {
        Motion.Move((-1 * Freeze));
        Motion.TurnRight(45);
        Motion.Move(Freeze);
        Freeze += (Motion.Direction() == 0);
    }
}
