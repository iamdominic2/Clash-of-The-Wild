//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    Looks.Hide();
}


//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(Game Over)
{
    Looks.Show();
    Looks.SetEffectTo(GHOST, 100);
    Repeat (20)
    {
        Looks.ChangeEffectBy(GHOST, -5);
    }
}
