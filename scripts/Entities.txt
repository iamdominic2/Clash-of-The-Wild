//----- Start as clone ------------------------------------------------------------------------------------------------

Control.WhenIStartAsClone()
{
    Forever
    {
        Looks.Say("");
        If Sensing.TouchingObject(Player)
        {
            If (type == "item")
            {
                If (price > 0)
                {
                    If (Not (List.ContainsItem(Text, 24)))
                    {
                        List.Add(Text, -170);
                        List.Add(Text, 160);
                        List.Add(Text, Operator.Join(Operator.Join("G", GOLD), Operator.Join("P", price)));
                        List.Add(Text, 24);
                    }
                }
                If ((((Sensing.KeyPressed(z) Or Sensing.KeyPressed(m)) Or Sensing.KeyPressed(space)) Or (STICK ACTION == 2)) Or (price < 1))
                {
                    GOLD += (-1 * price);
                    If (GOLD < 0)
                    {
                        GOLD += price;
                        Sound.PlayUntilDone(Buzz Whir);
                        Repeat (5)
                        {
                            Control.Wait(0);
                        }
                    }
                    Else
                    {
                        If (Costume.GetNumberName(number) == 8)
                        {
                            Sound.Play(Collect);
                            List.Add(Flags, flag);
                            PLAYER HEALTH += 3;
                            PLAYER MAX HP += 3;
                            List.Add(Text, Sensing.Of(x position, Player));
                            List.Add(Text, Sensing.Of(y position, Player));
                            List.Add(Text, "U");
                            List.Add(Text, 46);
                            Control.DeleteThisClone();
                        }
                        Else
                        {
                            Sound.Play(Coin);
                            List.Add(Flags, flag);
                            List.Add(Inventory, Costume.GetNumberName(name));
                            Repeat (9)
                            {
                                Looks.ChangeEffectBy(GHOST, 20);
                                If (Costume.GetNumberName(name) == "coin")
                                {
                                    List.Add(Text, -170);
                                    List.Add(Text, 160);
                                    List.Add(Text, Operator.Join("G", GOLD));
                                    List.Add(Text, 24);
                                }
                                If (price > 0)
                                {
                                    List.Add(Text, -170);
                                    List.Add(Text, 160);
                                    List.Add(Text, Operator.Join("G", GOLD));
                                    List.Add(Text, 24);
                                }
                            }
                            Control.DeleteThisClone();
                        }
                    }
                }
            }
            Else
            {
                Looks.Say(speech);
            }
        }
    }
}


//----- Custom blocks -------------------------------------------------------------------------------------------------

Define Spawn(string Item) Price(string Price) Scene(string text) Flag(string flag) (warp=false)
{
    If ((Sensing.Of(costume name, Wall) == text) And (Not (List.ContainsItem(Flags, flag))))
    {
        Looks.SwitchCostumeTo(Item);
        type = "item";
        price = Price;
        flag = flag;
        Control.CreateCloneOf(_myself_);
    }
}

Define Spawn character(string type) Say:(string text) Scene(string scene) Flag(string flag) (warp=false)
{
    If ((Sensing.Of(costume name, Wall) == scene) And (Not (List.ContainsItem(Flags, flag))))
    {
        Looks.SwitchCostumeTo(type);
        speech = text;
        type = "Character";
        Control.CreateCloneOf(_myself_);
    }
}
