//----- Start as clone ------------------------------------------------------------------------------------------------

Control.WhenIStartAsClone()
{
    Forever
    {
        If (Not (List.ContainsItem(Nopoint, Costume.GetNumberName(number))))
        {
            Motion.PointTowards(Player);
        }
        If (Sensing.TouchingObject(_edge_) And List.ContainsItem(Nopoint, Costume.GetNumberName(number)))
        {
            Control.DeleteThisClone();
        }
        Motion.ChangeXBy((sin(Motion.Direction()) * speed));
        If ((Sensing.TouchingObject(Wall) Or Sensing.TouchingColour(#66655e)) And (Not (List.ContainsItem(Nopoint, Costume.GetNumberName(number)))))
        {
            Motion.ChangeXBy((-1 * (sin(Motion.Direction()) * speed)));
        }
        Motion.ChangeYBy((cos(Motion.Direction()) * speed));
        If ((Sensing.TouchingObject(Wall) Or Sensing.TouchingColour(#66655e)) And (Not (List.ContainsItem(Nopoint, Costume.GetNumberName(number)))))
        {
            Motion.ChangeYBy((-1 * (cos(Motion.Direction()) * speed)));
        }
        roll = Operator.Random(0.85 , 1.15);
        Call Collision;
        If ((Operator.Random(1 , 50) == 1) And (Costume.GetNumberName(number) == 8))
        {
            Repeat (12)
            {
                Call Collision;
            }
            Motion.SetRotationStyle(all around);
            Motion.Move(35);
            orghp = Health;
            Call Spawn("Fireball") Speed(3.6) Scene("") Dmg(10) Hp(4) Defense(-21) Flag("");
            Looks.SwitchCostumeTo(Fireguy);
            Motion.SetRotationStyle(left-right);
            Motion.Move(-35);
            speed = 1;
            dmg = 12;
            Health = orghp;
            def = 4;
        }
        If ((Operator.Random(1 , (69 / (((Scene X == -10) * 2) + 1))) == 1) And (Costume.GetNumberName(number) == 11))
        {
            Repeat (16)
            {
                Call Collision;
            }
            Motion.SetRotationStyle(all around);
            Motion.Move(35);
            orghp = Health;
            Call Spawn("Arrow") Speed((((Scene X == -10) * 2) + 4)) Scene("") Dmg(14) Hp(4) Defense(-13) Flag("");
            Looks.SwitchCostumeTo(Shooter);
            Motion.SetRotationStyle(left-right);
            Motion.Move(-35);
            speed = 1;
            dmg = 7;
            Health = orghp;
            def = 0;
        }
    }
}

Define Die (warp=false)
{
    If (Health < 1)
    {
        Sound.Play(Bonk);
        Looks.Hide();
        Repeat (4)
        {
            Control.Wait(0);
        }
        List.Add(Text, Motion.XPosition());
        List.Add(Text, Motion.YPosition());
        If (Exp.Extra[Costume.GetNumberName(number)] > -50)
        {
            List.Add(Text, Operator.Round((((((Operator.Round(((((dmg * 1.2) + (sqrt(speed) * 1.5)) + ((def + 7) / 2)) * 1.4)) + (Exp.Extra[Costume.GetNumberName(number)] * 1.4)) + ((Looks.Size() / 5) - 16)) * ((((Scene X < 0) + (((Scene X < -4) * 1.5) + ((Scene X < -8) * 3.5))) / 1.5) + 1)) / 5.5) * 5)));
        }
        Else
        {
            List.Add(Text, 1);
        }
        List.Add(Text, 121);
        If (Costume.GetNumberName(number) == 5)
        {
            List.DeleteItem(Flags, List.IndexOf(Flags, "unspider"));
            List.Add(Flags, "spiderdefeated");
        }
        If (Costume.GetNumberName(number) == 13)
        {
            List.Add(Flags, "wizardgone");
        }
        If (Costume.GetNumberName(number) == 14)
        {
            List.Add(Flags, "win");
        }
        Repeat (5)
        {
            If (Exp.Extra[Costume.GetNumberName(number)] > -100)
            {
                EXP += ((((Operator.Round(((((dmg * 1.2) + (sqrt(speed) * 1.5)) + ((def + 7) / 2)) * 1.4)) + (Exp.Extra[Costume.GetNumberName(number)] * 1.4)) + ((Looks.Size() / 5) - 16)) * ((((Scene X < 0) + (((Scene X < -4) * 1.5) + ((Scene X < -8) * 3.5))) / 1.5) + 1)) / 5.5);
            }
            Else
            {
                EXP += 0.2;
            }
        }
        Control.DeleteThisClone();
    }
    Looks.Hide();
    Control.Wait(0);
    Looks.Show();
    Wait Until (Not (Sensing.TouchingObject(Weapon)));
}
