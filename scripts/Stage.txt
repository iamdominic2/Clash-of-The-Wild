//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    Forever
    {
        Looks.SwitchCostumeTo(Operator.Join(Operator.Join("x", Scene X), Operator.Join("y", Scene Y)));
    }
}

WhenGreenFlagClicked()
{
    Sound.SetVolumeTo(100);
    Music = 1;
    Forever
    {
        Sound.PlayUntilDone(Music);
    }
}

WhenGreenFlagClicked()
{
    Control.Wait(0);
    Forever
    {
        Wait Until (EXP > (Level Curve[LEVEL] + 1));
        EXP += (0 - Level Curve[LEVEL]);
        LEVEL += 1;
        PLAYER HEALTH += 1;
        PLAYER MAX HP += 1;
        Control.Wait(0.2);
        Sound.Play(Level Up);
        List.Add(Text, Sensing.Of(x position, Player));
        List.Add(Text, Sensing.Of(y position, Player));
        List.Add(Text, "L");
        List.Add(Text, 55);
    }
}

WhenGreenFlagClicked()
{
    Forever
    {
        If (PLAYER HEALTH < (PLAYER MAX HP / 3))
        {
            Sound.PlayUntilDone(pokmon_low_hp_sound_no_battle_the_(getmp3);
        }
    }
}


//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(Game Over)
{

}
