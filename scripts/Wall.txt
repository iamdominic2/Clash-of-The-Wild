//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    List.DeleteAll(Inventory);
    Forever
    {
        Looks.ClearGraphicEffects();
        Pen.Clear();
        Looks.SetSizeTo(100);
        Looks.SwitchCostumeTo(Operator.Join(Operator.Join("x", Scene X), Operator.Join("y", Scene Y)));
        Looks.SetEffectTo(BRIGHTNESS, -95);
        If List.ContainsItem(Flags, "spiderdefeated")
        {
            Looks.SwitchCostumeTo(Operator.Join(Operator.Join(Operator.Join("x", Scene X), Operator.Join("y", Scene Y)), "_e"));
        }
        If List.ContainsItem(Flags, "wizardgone")
        {
            Looks.SwitchCostumeTo(Operator.Join(Operator.Join(Operator.Join("x", Scene X), Operator.Join("y", Scene Y)), "_w"));
        }
        If List.ContainsItem(Flags, "win")
        {
            Looks.SwitchCostumeTo(Operator.Join(Operator.Join(Operator.Join("x", Scene X), Operator.Join("y", Scene Y)), "s"));
        }
        If (Not ((Costume.GetNumberName(number) == 25)))
        {
            Call Pen Effect;
        }
        Looks.ClearGraphicEffects();
        Pen.Stamp();
        Looks.SetEffectTo(GHOST, 100);
    }
}

//----- Custom blocks -------------------------------------------------------------------------------------------------

Define Pen Effect (warp=true)
{
    Move = 0;
    Motion.PointInDirection(90);
    Motion.GoToXY(0, 0);
    Repeat (3)
    {
        Move += 1;
        Repeat (16)
        {
            Motion.Move(Move);
            Pen.Stamp();
            Motion.Move((0 - Move));
            Motion.TurnRight(22.5);
        }
    }
}

Define Switch song(string n) name(string nm) (warp=false)
{
    If ((Costume.GetNumberName(name) == nm) And (Not ((n == Music))))
    {
        Music = n;
        Sound.StopAllSounds();
    }
}
